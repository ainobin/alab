Experiment 20: Write a C++ program to implement Breadth First Search algorithm
----------------------------------------------------
#include <iostream>
#include <queue>
#include <vector>
using namespace std;

int main() {
    int n = 6;
    vector<vector<int>> graph = {
        {1, 2}, {0, 3, 4}, {0, 4}, {1, 5}, {1, 2, 5}, {3, 4}
    };
    vector<bool> visited(n, false);
    queue<int> q;

    q.push(0);
    visited[0] = true;

    cout << "BFS Traversal: ";
    while (!q.empty()) {
        int node = q.front();
        q.pop();
        cout << node << " ";

        for (int neighbor : graph[node]) {
            if (!visited[neighbor]) {
                visited[neighbor] = true;
                q.push(neighbor);
            }
        }
    }
    return 0;
}

Experiment 21: Write a C++ program to implement Depth First Search algorithm
----------------------------------------------------
#include <iostream>
#include <vector>
using namespace std;

vector<vector<int>> graph;
vector<bool> visited;

void dfs(int node) {
    visited[node] = true;
    cout << node << " ";

    for (int neighbor : graph[node]) {
        if (!visited[neighbor]) {
            dfs(neighbor);
        }
    }
}

int main() {
    int n = 6;
    graph = {
        {1, 2}, {0, 3, 4}, {0, 4}, {1, 5}, {1, 2, 5}, {3, 4}
    };
    visited.assign(n, false);

    cout << "DFS Traversal: ";
    dfs(0);
    return 0;
}

Experiment 22: Write a C++ program to implement A* Search algorithm.
#include <iostream>
#include <vector>
#include <queue>
using namespace std;

struct Node
{
    int id;     // Current node
    int cost;   // f = g + h

    bool operator>(const Node &other) const
    {
        return cost > other.cost;
    }
};

int main()
{
    // Graph: {neighbor, edge cost}
    vector<vector<pair<int,int>>> graph =
    {
        {{1,2}, {2,5}},   // 0
        {{2,4}, {3,6}},   // 1
        {{3,2}},          // 2
        {{4,3}},          // 3
        {}                // 4
    };

    vector<int> h = {7,6,2,1,0};

    priority_queue<Node, vector<Node>, greater<Node>> pq;

    vector<int> g(5, 100000);
    vector<int> parent(5, -1);

    int start = 0;
    int goal = 4;

    g[start] = 0;
    pq.push({start, h[start]});

    while (!pq.empty())
    {
        Node current = pq.top();
        pq.pop();

        if (current.id == goal)
            break;

        for (auto edge : graph[current.id])
        {
            int next = edge.first;
            int weight = edge.second;

            if (g[current.id] + weight < g[next])
            {
                g[next] = g[current.id] + weight;
                parent[next] = current.id;

                int f = g[next] + h[next];
                pq.push({next, f});
            }
        }
    }

    cout << "Cost = " << g[goal] << endl;

    cout << "Path: ";

    vector<int> path;
    for (int i = goal; i != -1; i = parent[i])
        path.push_back(i);

    for (int i = path.size()-1; i >= 0; i--)
        cout << path[i] << " ";
}
